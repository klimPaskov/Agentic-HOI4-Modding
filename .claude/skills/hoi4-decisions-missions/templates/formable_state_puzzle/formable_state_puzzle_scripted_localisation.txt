# Fixed three-state count example. Generate one descending text branch per
# possible count, and include every manifest state wrapper in each count_triggers
# block. The first matching threshold is the exact live numerator.
defined_text = {
	name = formable_<FORMABLE_ID>_qualifying_count_text
	text = {
		trigger = {
			count_triggers = {
				amount > 2
				formable_<FORMABLE_ID>_state_<STATE_ID_A>_qualifies = yes
				formable_<FORMABLE_ID>_state_<STATE_ID_B>_qualifies = yes
				formable_<FORMABLE_ID>_state_<STATE_ID_C>_qualifies = yes
			}
		}
		localization_key = "<FORMABLE_ID>_qualifying_count_3"
	}
	text = {
		trigger = {
			count_triggers = {
				amount > 1
				formable_<FORMABLE_ID>_state_<STATE_ID_A>_qualifies = yes
				formable_<FORMABLE_ID>_state_<STATE_ID_B>_qualifies = yes
				formable_<FORMABLE_ID>_state_<STATE_ID_C>_qualifies = yes
			}
		}
		localization_key = "<FORMABLE_ID>_qualifying_count_2"
	}
	text = {
		trigger = {
			count_triggers = {
				amount > 0
				formable_<FORMABLE_ID>_state_<STATE_ID_A>_qualifies = yes
				formable_<FORMABLE_ID>_state_<STATE_ID_B>_qualifies = yes
				formable_<FORMABLE_ID>_state_<STATE_ID_C>_qualifies = yes
			}
		}
		localization_key = "<FORMABLE_ID>_qualifying_count_1"
	}
	text = {
		localization_key = "<FORMABLE_ID>_qualifying_count_0"
	}
}
